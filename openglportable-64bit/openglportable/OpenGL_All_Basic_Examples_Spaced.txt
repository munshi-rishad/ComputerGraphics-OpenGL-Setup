6.  CIRCLE DRAWING

#include <math.h>

void circle(float r) { glBegin(GL_POLYGON);

    for(float i=0;i<360;i++)
    {
        float a=i*3.1416/180;

        glVertex2f(r*cos(a),r*sin(a));
    }

    glEnd();

}

12. DDA LINE ALGORITHM

dx=x2-x1;

dy=y2-y1;

steps=max(abs(dx),abs(dy));

xinc=dx/steps;

yinc=dy/steps;

for(i=0;i<=steps;i++) { glVertex2i(x,y);

    x=x+xinc;

    y=y+yinc;

}

13. BRESENHAM LINE ALGORITHM

dx=x2-x1;

dy=y2-y1;

p=2*dy-dx;

while(x<x2) { if(p>=0) { y++;

        p=p+2*dy-2*dx;
    }

    else
    {
        p=p+2*dy;
    }

    x++;

}

14. BASIC OPENGL MAIN STRUCTURE

    drawing code

    glutInitWindowSize(500,500);

    glutCreateWindow("OpenGL");
